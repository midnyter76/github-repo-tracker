Purpose: `reports/*.html` is 86% of the directory bulk (~78KB/day, ~28MB/year) and is a

~11KB/day and is explicitly NOT pruned.

Output: a `REPORT_HTML_RETENTION_DAYS` constant, a `prune_reports()` sharing
`prune_snapshots()`'s loop via a private `_prune_dated()` helper, a collector call in the
existing prune-LAST section, and a widened deletion-staging step in `daily.yml` —
without which the whole fix is inert (files are deleted on the ephemeral runner but never
recorded as deletions in the commit).

def prune_snapshots(
    now: datetime,
    snapshots_dir: Path = SNAPSHOTS_DIR,
    retention_days: int = SNAPSHOT_RETENTION_DAYS,
) -> list[Path]:
    if not snapshots_dir.exists():
        return []
    cutoff = (now - timedelta(days=retention_days)).date()
    pruned: list[Path] = []
    for snap_path in snapshots_dir.glob("*.json"):
        try:
            snap_date = date.fromisoformat(snap_path.stem)
        except ValueError:
            continue  # ignore non-date-named files (e.g. backup.json)
        if snap_date < cutoff:
            snap_path.unlink()
            pruned.append(snap_path)
    return pruned
```

def _prune_dated(
    now: datetime,
    directory: Path,
    retention_days: int,
    pattern: str,
) -> list[Path]:
    """Delete files in directory matching pattern whose filename stem is a date
    older than retention_days. Shared body of prune_snapshots + prune_reports.

    Pruning is by filename stem date, never mtime — mtime is meaningless on a
    fresh GitHub Actions checkout. Non-date-named files are silently ignored.
    Safe to call when directory does not exist — returns [].
    """
    if not directory.exists():
        return []

    cutoff = (now - timedelta(days=retention_days)).date()
    pruned: list[Path] = []

    for path in directory.glob(pattern):
        try:
            stem_date = date.fromisoformat(path.stem)
        except ValueError:
            continue  # ignore non-date-named files (e.g. backup.json, notes.html)

        if stem_date < cutoff:
            path.unlink()
            pruned.append(path)

    return pruned
```
